async function autoFindSetlist() { const datumVal = document.getElementById('DATUM').value.trim(); const cityVal = document.getElementById('MESTO').value.trim(); if (!datumVal) { showStatus('Please enter a valid Date (YYYY-MM-DD) first.', 'red'); return; } const parts = datumVal.split('-'); if (parts.length !== 3 || parts[0].length !== 4) { showStatus(`Invalid date format '${datumVal}'. Expected YYYY-MM-DD.`, 'red'); return; } const concertYear = parts[0]; const apiFormattedDate = `${parts[2]}-${parts[1]}-${parts[0]}`; // DD-MM-YYYY showStatus(`Searching Setlist.fm for Joe Jackson show on ${apiFormattedDate}...`, 'blue'); try { let foundShow = null; // 1. Zkusíme přímé vyhledání podle data (ignorujeme případnou 404) const primaryApiUrl = `https://api.setlist.fm/rest/1.0/artist/${JOE_JACKSON_MBID}/setlists?date=${apiFormattedDate}`; try { const primaryData = await fetchSetlistApiSecure(primaryApiUrl); if (primaryData && primaryData.setlist) { const list = Array.isArray(primaryData.setlist) ? primaryData.setlist : [primaryData.setlist]; if (list.length > 0) foundShow = list[0]; } } catch (e) { // Pokud API vrátí 404 (Not Found), plynule pokračujeme na záložní hledání podle roku console.log('Direct date lookup returned 404, falling back to year search...'); } // 2. Záložní vyhledávání podle celého ROKU (Setlist.fm 404 obchvat) if (!foundShow) { showStatus(`Searching Setlist.fm archive for year ${concertYear}...`, 'blue'); const fallbackApiUrl = `https://api.setlist.fm/rest/1.0/artist/${JOE_JACKSON_MBID}/setlists?year=${concertYear}`; const fallbackData = await fetchSetlistApiSecure(fallbackApiUrl); if (fallbackData && fallbackData.setlist) { const list = Array.isArray(fallbackData.setlist) ? fallbackData.setlist : [fallbackData.setlist]; // Najdeme koncert podle přesného data události (eventDate na Setlist.fm je ve formátu DD-MM-YYYY) foundShow = list.find(s => { if (s.eventDate === apiFormattedDate) return true; if (cityVal && s.venue && s.venue.city) { return s.venue.city.name.toLowerCase().includes(cityVal.toLowerCase()); } return false; }); } } if (!foundShow) { showStatus(`No show recorded on Setlist.fm for date ${apiFormattedDate}${cityVal ? ' in ' + cityVal : ''}.`, 'red'); return; } if (foundShow.url) { document.getElementById('SETLIST_URL').value = foundShow.url; } const songNames = parseSongsFromShow(foundShow); if (songNames.length === 0) { showStatus(`Found concert on Setlist.fm (${foundShow.url}), but no songs are logged for this show. Direct URL saved!`, 'blue'); return; } document.getElementById('SETLIST').value = songNames.join(', '); recalculateSongCount(); showStatus(`✅ Successfully loaded ${songNames.length} songs and updated URL from Setlist.fm!`, 'green'); } catch (err) { showStatus(`❌ ${err.message}`, 'red'); } }
